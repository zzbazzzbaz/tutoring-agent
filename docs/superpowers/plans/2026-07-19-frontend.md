# 教学 Agent 前端 Implementation Plan

> **For agentic workers:** REQUIRED SUB-SKILL: Use superpowers:subagent-driven-development (recommended) or superpowers:executing-plans to implement this plan task-by-task. Steps use checkbox (`- [ ]`) syntax for tracking.

**Goal:** 实现教学 Agent 前端（学生端 + 教师端），与后端 FastAPI + LightRAG + LangChain 1.x 协同。

**Architecture:** 单 Vite 应用 + 角色路由（`/student/*`、`/teacher/*`）。共享层（api/stores/components/ui/lib/types）+ 角色隔离的 features。教师端知识库/图谱拷贝适配自 LightRAG webui 的 DocumentManager + GraphViewer。

**Tech Stack:** React 19 + TypeScript + Vite 8 + Bun + Tailwind v4 + Radix UI (shadcn/ui) + Zustand + axios + react-router 7 + Sigma.js/graphology + react-markdown + remark-gfm + rehype-katex + lucide-react + sonner。无 i18next，无 TanStack Query，无前端单测。

**Spec:** `docs/superpowers/specs/2026-07-19-frontend-design.md`

## Global Constraints

- 包管理器：Bun（所有 `bun install` / `bun run` / `bunx`）
- 工作目录：`frontend/`（所有前端命令在此目录运行）
- 后端地址：`http://localhost:8000`（dev proxy 转发）
- 前端 dev 端口：`5173`
- TypeScript 严格模式，路径别名 `@/* -> src/*`
- `@typescript-eslint/no-explicit-any` 允许（与 LightRAG webui 一致）
- 不写前端单元测试（验证靠 `bun run lint` + `bun run build` + 手动浏览器 + agent-browser E2E）
- 中文 UI（无 i18next，直接硬编码中文文案）
- 后端错误响应格式：`{"detail": {"error": {"code": "...", "message": "..."}}}`
- SSE 事件协议：`event: text | quiz_interrupt | done | error`，帧格式 `event: <name>\ndata: <json>\n\n`
- 后端契约参考：`backend/src/tutoring/*/router.py` + `backend/src/tutoring/*/schemas.py` + `backend/src/tutoring/agent/runtime.py`
- LightRAG webui 参考路径：`docs/_chat/LightRAG/lightrag_webui/`

```typescript
export function formatRelativeTime(iso: string): string {
  const date = new Date(iso)
  const diff = Date.now() - date.getTime()
  const sec = Math.floor(diff / 1000)
  const min = Math.floor(sec / 60)
  const hour = Math.floor(min / 60)
  const day = Math.floor(hour / 24)

  if (sec < 60) return '刚刚'
  if (min < 60) return `${min} 分钟前`
  if (hour < 24) return `${hour} 小时前`
  if (day < 7) return `${day} 天前`
  return date.toLocaleDateString('zh-CN')
}

export function formatDateTime(iso: string): string {
  return new Date(iso).toLocaleString('zh-CN', {
    year: 'numeric', month: '2-digit', day: '2-digit',
    hour: '2-digit', minute: '2-digit',
  })
}
```

```typescript
import { useEffect, useState, useCallback } from 'react'

interface UseAsyncResult<T> {
  data: T | null
  error: Error | null
  loading: boolean
  refetch: () => void
}

export function useAsync<T>(fn: () => Promise<T>, deps: unknown[] = []): UseAsyncResult<T> {
  const [data, setData] = useState<T | null>(null)
  const [error, setError] = useState<Error | null>(null)
  const [loading, setLoading] = useState(true)
  const [nonce, setNonce] = useState(0)

  const refetch = useCallback(() => setNonce(n => n + 1), [])

  useEffect(() => {
    let cancelled = false
    setLoading(true)
    fn()
      .then(d => { if (!cancelled) { setData(d); setError(null) } })
      .catch(e => { if (!cancelled) setError(e) })
      .finally(() => { if (!cancelled) setLoading(false) })
    return () => { cancelled = true }
    // eslint-disable-next-line react-hooks/exhaustive-deps
  }, [...deps, nonce])

  return { data, error, loading, refetch }
}
```

- [ ] **Step 6: 写 `src/lib/parseSSE.ts`**

```typescript
export interface SSEEvent {
  event: string
  data: any
}

/**
 * 解析 SSE 流。后端帧格式：event: <name>\ndata: <json>\n\n
 */
export async function* parseSSE(response: Response): AsyncGenerator<SSEEvent> {
  const reader = response.body!.getReader()
  const decoder = new TextDecoder()
  let buffer = ''

  while (true) {
    const { done, value } = await reader.read()
    if (done) break
    buffer += decoder.decode(value, { stream: true })

    let idx: number
    while ((idx = buffer.indexOf('\n\n')) !== -1) {
      const frame = buffer.slice(0, idx)
      buffer = buffer.slice(idx + 2)

      const lines = frame.split('\n')
      const event = lines.find(l => l.startsWith('event: '))?.slice(7) || 'message'
      const dataLine = lines.find(l => l.startsWith('data: '))?.slice(6) || '{}'
      try {
        yield { event, data: JSON.parse(dataLine) }
      } catch {
        yield { event, data: { raw: dataLine } }
      }
    }
  }
}
```

```typescript
  role: Role | null

      isAuthenticated: false,
      username: null,

        set({
```

```typescript
  const username = useAuthStore.getState().username
  if (username) {
    config.headers['X-User-Id'] = username
  }
  return config
})

// 响应拦截器：统一错误 toast
apiClient.interceptors.response.use(
  res => res,
  err => {
    const detail = err.response?.data?.detail?.error
    if (detail) {
      toast.error(`${detail.code}: ${detail.message}`)
    } else if (err.response) {
      toast.error(`HTTP ${err.response.status}: ${err.message}`)
    } else {
      toast.error(err.message || '网络错误')
    }
    return Promise.reject(err)
  },
)

// SSE 流式请求用的 header 注入（fetch 不走 axios 拦截器）
export function getAuthHeaders(): Record<string, string> {
  const username = useAuthStore.getState().username
  return username ? { 'X-User-Id': username } : {}
}
```

```typescript
import { getAuthHeaders } from './client'
import { parseSSE, type SSEEvent } from '@/lib/parseSSE'

const API_BASE = import.meta.env.VITE_API_BASE || ''

export async function* streamMessage(
  sessionId: string,
  content: string,
  signal?: AbortSignal,
): AsyncGenerator<SSEEvent> {
  const res = await fetch(`${API_BASE}/chat/sessions/${sessionId}/messages`, {
    method: 'POST',
    headers: {
      'Content-Type': 'application/json',
      ...getAuthHeaders(),
    },
    body: JSON.stringify({ content }),
    signal,
  })
  if (!res.ok) {
    const text = await res.text().catch(() => '')
    throw new Error(`HTTP ${res.status}: ${text}`)
  }
  yield* parseSSE(res)
}

export async function* resumeQuiz(
  sessionId: string,
  action: 'answer' | 'skip',
  answer: string | null,
  signal?: AbortSignal,
): AsyncGenerator<SSEEvent> {
  const body: Record<string, unknown> = { action }
  if (action === 'answer' && answer !== null) {
    body.answer = answer
  }
  const res = await fetch(`${API_BASE}/chat/sessions/${sessionId}/quiz/resume`, {
    method: 'POST',
    headers: {
      'Content-Type': 'application/json',
      ...getAuthHeaders(),
    },
    body: JSON.stringify(body),
    signal,
  })
  if (!res.ok) {
    const text = await res.text().catch(() => '')
    throw new Error(`HTTP ${res.status}: ${text}`)
  }
  yield* parseSSE(res)
}
```

```typescript
import { create } from 'zustand'
import { toast } from 'sonner'
import { sessionsApi } from '@/api/sessions'
import { streamMessage, resumeQuiz } from '@/api/chat'
import type { SessionOut } from '@/types/session'
import type { QuestionOut } from '@/types/question'
import type { ChatItem, StreamStatus } from '@/types/chat'

interface ChatState {
  currentSession: SessionOut | null
  items: ChatItem[]
  streamStatus: StreamStatus
  pendingQuiz: QuestionOut | null
  streamingAgentContent: string
  lastError: { message: string; code: string } | null
  currentAbort: AbortController | null
  loadingSession: boolean

  loadSession: (id: string) => Promise<void>
  sendMessage: (content: string) => Promise<void>
  submitQuiz: (action: 'answer' | 'skip', answer?: string) => Promise<void>
  resetStream: () => void
  clearCurrent: () => void
}

export const useChatStore = create<ChatState>((set, get) => ({
  currentSession: null,
  items: [],
  streamStatus: 'idle',
  pendingQuiz: null,
  streamingAgentContent: '',
  lastError: null,
  currentAbort: null,
  loadingSession: false,

  loadSession: async (id) => {
    set({ loadingSession: true })
    try {
      const [session, messages, questions] = await Promise.all([
        sessionsApi.get(id),
        sessionsApi.listMessages(id),
        sessionsApi.listQuestions(id),
      ])
      const items: ChatItem[] = [
        ...messages.map(m => ({ type: 'message' as const, ...m })),
        ...questions.map(q => ({ type: 'question' as const, question: q })),
      ].sort((a, b) => +new Date(a.created_at) - +new Date(b.created_at))

      set({
        currentSession: session, items,
        streamStatus: 'idle', pendingQuiz: null,
        streamingAgentContent: '', lastError: null,
      })
    } finally {
      set({ loadingSession: false })
    }
  },

  sendMessage: async (content) => {
    const { currentSession } = get()
    if (!currentSession) return
    if (get().streamStatus === 'streaming' || get().streamStatus === 'quiz_interrupted') return

    const userMsg = {
      type: 'message' as const,
      role: 'user' as const,
      content,
      created_at: new Date().toISOString(),
    }
    set(state => ({
      items: [...state.items, userMsg],
      streamStatus: 'streaming' as const,
      streamingAgentContent: '',
      lastError: null,
    }))

    const ac = new AbortController()
    set({ currentAbort: ac })

    try {
      for await (const ev of streamMessage(currentSession.id, content, ac.signal)) {
        switch (ev.event) {
          case 'text':
            set(state => ({ streamingAgentContent: state.streamingAgentContent + ev.data.content }))
            break
          case 'quiz_interrupt':
            set({ pendingQuiz: ev.data, streamStatus: 'quiz_interrupted' })
            break
          case 'done':
            if (get().streamStatus === 'streaming') {
              const agentMsg = {
                type: 'message' as const,
                role: 'agent' as const,
                content: get().streamingAgentContent,
                created_at: new Date().toISOString(),
              }
              set(state => ({
                items: [...state.items, agentMsg],
                streamStatus: 'idle' as const,
                streamingAgentContent: '',
              }))
            } else if (get().streamStatus === 'quiz_interrupted') {
              set({ streamingAgentContent: '' })
            }
            break
          case 'error':
            set({ streamStatus: 'error', lastError: ev.data })
            toast.error(ev.data.message)
            set(state => ({ items: state.items.filter(it => it !== userMsg) }))
            break
        }
      }
      if (get().streamStatus === 'streaming') {
        set({ streamStatus: 'error', lastError: { message: '连接中断', code: 'STREAM_CLOSED' } })
        toast.error('连接中断')
        set(state => ({ items: state.items.filter(it => it !== userMsg) }))
      }
    } catch (e: any) {
      if (e.name !== 'AbortError') {
        set({ streamStatus: 'error', lastError: { message: String(e), code: 'NETWORK' } })
        toast.error('网络错误')
        set(state => ({ items: state.items.filter(it => it !== userMsg) }))
      }
    } finally {
      set({ currentAbort: null })
    }
  },

  submitQuiz: async (_action, _answer) => {
    // Task 9 实现
  },

  resetStream: () => {
    const ac = get().currentAbort
    if (ac) ac.abort()
    set({
      streamStatus: 'idle',
      streamingAgentContent: '',
      pendingQuiz: null,
      lastError: null,
      currentAbort: null,
    })
  },

  clearCurrent: () => {
    get().resetStream()
    set({ currentSession: null, items: [], loadingSession: false })
  },
}))
```

```typescript
import { useState, useRef, useEffect } from 'react'
import { useChatStore } from '@/stores/chat'
import { Button } from '@/components/ui/button'
import { Send } from 'lucide-react'

const MAX_CHARS = 2000

export default function ChatInput() {
  const [content, setContent] = useState('')
  const textareaRef = useRef<HTMLTextAreaElement>(null)
  const streamStatus = useChatStore(s => s.streamStatus)
  const sendMessage = useChatStore(s => s.sendMessage)

  const disabled = streamStatus === 'streaming' || streamStatus === 'quiz_interrupted'
  const placeholder = streamStatus === 'quiz_interrupted'
    ? '请先答题或跳过当前题目'
    : streamStatus === 'streaming'
      ? '生成中...'
      : '输入消息，Enter 发送'

  const handleSubmit = async () => {
    const trimmed = content.trim()
    if (!trimmed || disabled) return
    setContent('')
    await sendMessage(trimmed)
  }

  const handleKeyDown = (e: React.KeyboardEvent<HTMLTextAreaElement>) => {
    if (e.key === 'Enter' && !e.shiftKey) {
      e.preventDefault()
      handleSubmit()
    }
  }

  useEffect(() => {
    const ta = textareaRef.current
    if (!ta) return
    ta.style.height = 'auto'
    ta.style.height = Math.min(ta.scrollHeight, 200) + 'px'
  }, [content])

  return (
    <div className="mx-auto max-w-3xl">
      <div className="flex items-end gap-2">
        <textarea
          ref={textareaRef}
          value={content}
          onChange={e => setContent(e.target.value.slice(0, MAX_CHARS))}
          onKeyDown={handleKeyDown}
          placeholder={placeholder}
          disabled={disabled}
          rows={1}
          className="flex-1 resize-none rounded-md border bg-background p-3 text-sm focus:outline-none focus:ring-1 focus:ring-ring disabled:opacity-50"
        />
        <Button onClick={handleSubmit} disabled={disabled || !content.trim()} size="icon">
          <Send className="h-4 w-4" />
        </Button>
      </div>
      <div className="mt-1 text-right text-xs text-muted-foreground">
        {content.length}/{MAX_CHARS}
        {streamStatus === 'streaming' && ' · 生成中...'}
      </div>
    </div>
  )
}
```

```typescript
interface Props {

  pending?: QuizInterruptData

export default function QuizCard({ question, pending }: Props) {
  const streamStatus = useChatStore(s => s.streamStatus)
  const submitQuiz = useChatStore(s => s.submitQuiz)

  const data = pending || question
  const isPending = !!pending && streamStatus === 'quiz_interrupted'
  const isLocked = !isPending || streamStatus === 'streaming'

  const qType = data?.q_type
  const options = data?.options
  const stem = data?.stem

  const [selected, setSelected] = useState<string>('')
  const [multiSelected, setMultiSelected] = useState<string[]>([])
  const [shortAnswer, setShortAnswer] = useState('')

  if (!data) return null

  const handleMultiChange = (v: string) => {
    setMultiSelected(prev => prev.includes(v) ? prev.filter(x => x !== v) : [...prev, v])
  }

  const handleSubmit = async () => {
    let answer = ''
    if (qType === 'single') answer = selected
    else if (qType === 'multi') answer = multiSelected.sort().join(',')
    else if (qType === 'short') answer = shortAnswer
    await submitQuiz('answer', answer)
  }

  const handleSkip = async () => {
    await submitQuiz('skip')
  }

  const verdict = question?.verdict
  const verdictVariant = verdict === 'correct' ? 'default' : verdict === 'partial' ? 'secondary' : 'destructive'
  const verdictText = verdict === 'correct' ? '✅ 答对'
    : verdict === 'partial' ? '🟡 部分对'
    : verdict === 'wrong' ? '❌ 答错'
    : question?.status === 'skipped' ? '⏭️ 已跳过' : ''

  return (
    <div className="rounded-lg border p-4">
      <div className="mb-3 flex items-center justify-between">
        <div className="flex items-center gap-2">

            {qType === 'single' ? '单选题' : qType === 'multi' ? '多选题' : '简答题'}
          </span>
          {(question?.topic || pending?.topic) && (
            <Badge variant="outline" className="text-xs">{question?.topic || pending?.topic}</Badge>
          )}
        </div>
        {verdictText && <Badge variant={verdictVariant}>{verdictText}</Badge>}
        {isPending && <Badge variant="outline">作答中</Badge>}
      </div>

      <div className="mb-3 text-sm">{stem}</div>

      {options && qType === 'single' && (
        <RadioGroup
          value={question?.user_answer || selected}
          onValueChange={setSelected}
          disabled={isLocked}
          className="space-y-2"
        >
          {options.map((opt, i) => {
            const letter = String.fromCharCode(65 + i)
            const isUserAnswer = question?.user_answer === opt
            const isCorrectAnswer = question?.answer?.includes(opt)
            return (
              <div key={i} className={cn(
                'flex items-center gap-2 rounded-md p-2',
                isLocked && isUserAnswer && 'bg-yellow-100 dark:bg-yellow-900/30',
                isLocked && isCorrectAnswer && 'bg-green-100 dark:bg-green-900/30',

                <label htmlFor={`q-${i}`} className="text-sm">

              </div>
            )
          })}
        </RadioGroup>

            const letter = String.fromCharCode(65 + i)
            const userAns = question?.user_answer?.split(',') || []
            const isUserAnswer = userAns.includes(opt)
            const isCorrectAnswer = question?.answer?.split(',').includes(opt)
            const checked = isLocked ? isUserAnswer : multiSelected.includes(opt)
            return (
              <div key={i} className={cn(
                'flex items-center gap-2 rounded-md p-2',
                isLocked && isUserAnswer && 'bg-yellow-100 dark:bg-yellow-900/30',
```

```typescript
import { create } from 'zustand'
import { documentsApi, type DocumentItem, type PipelineStatus } from '@/api/documents'

interface DocumentsState {
  documents: DocumentItem[]
  total: number
  page: number
  pageSize: number
  pipelineStatus: PipelineStatus | null
  loading: boolean
  pollTimer: ReturnType<typeof setInterval> | null

  fetchPage: (page?: number) => Promise<void>
  startPolling: () => void
  stopPolling: () => void
  uploadFile: (file: File) => Promise<void>
  insertText: (text: string) => Promise<void>
  scan: () => Promise<void>
  deleteDoc: (docId: string) => Promise<void>
}

export const useDocumentsStore = create<DocumentsState>((set, get) => ({
  documents: [],
  total: 0,
  page: 1,
  pageSize: 20,
  pipelineStatus: null,
  loading: false,
  pollTimer: null,

  fetchPage: async (page) => {
    const p = page ?? get().page
    set({ loading: true })
    try {
      const res = await documentsApi.listPaginated({ page: p, page_size: get().pageSize })
      set({ documents: res.documents || [], total: res.total_count || 0, page: p })
    } finally {
      set({ loading: false })
    }
  },

  startPolling: () => {
    if (get().pollTimer) return
    const tick = async () => {
      try {
        const status = await documentsApi.pipelineStatus()
        set({ pipelineStatus: status })
        if (status.busy) {
          get().fetchPage()
        }
      } catch {
        // 忽略轮询错误
      }
    }
    tick()
    const timer = setInterval(tick, 5000)
    set({ pollTimer: timer })
  },

  stopPolling: () => {
    const timer = get().pollTimer
    if (timer) clearInterval(timer)
    set({ pollTimer: null })
  },

  uploadFile: async (file) => {
    await documentsApi.uploadFile(file)
    await get().fetchPage()
  },

  insertText: async (text) => {
    await documentsApi.insertText({ text })
    await get().fetchPage()
  },

  scan: async () => {
    await documentsApi.scan()
    await get().fetchPage()
  },

  deleteDoc: async (docId) => {
    await documentsApi.deleteDocument(docId)
    await get().fetchPage()
  },
}))
```

```typescript
  [key: string]: unknown

  },

  popularLabels: async (limit = 300): Promise<string[]> => {
    const res = await apiClient.get<string[]>('/graph/label/popular', { params: { limit } })
    return res.data
  },

  searchLabels: async (query: string): Promise<string[]> => {
    const res = await apiClient.get<string[]>('/graph/label/search', { params: { query } })
    return res.data
  },

  getGraph: async (params: { label: string; max_depth?: number; max_nodes?: number }): Promise<GraphData> => {
    const res = await apiClient.get<GraphData>('/graphs', {
      params: {
        label: params.label,
        max_depth: params.max_depth ?? 3,
        max_nodes: params.max_nodes ?? 1000,
      },
    })
```

```typescript
import { useEffect, useState } from 'react'
import { useNavigate, useParams } from 'react-router-dom'
import { toast } from 'sonner'
import { projectsApi } from '@/api/projects'
import MarkdownRenderer from '@/features/shared/MarkdownRenderer'
import { Button } from '@/components/ui/button'
import { Input } from '@/components/ui/input'
import { Label } from '@/components/ui/label'
import {
  Select, SelectContent, SelectItem, SelectTrigger, SelectValue,
} from '@/components/ui/select'
import type { ProjectOut } from '@/types/project'

const MAX_NAME = 200

export default function ProjectEditor() {
  const navigate = useNavigate()
  const { id } = useParams<{ id: string }>()
  const isEdit = !!id

  const [name, setName] = useState('')
  const [status, setStatus] = useState<'draft' | 'published'>('draft')
  const [backgroundMd, setBackgroundMd] = useState('')
  const [loading, setLoading] = useState(isEdit)
  const [saving, setSaving] = useState(false)

  useEffect(() => {
    if (!id) return
    projectsApi.get(id)
      .then((p: ProjectOut) => {
        setName(p.name)
        setStatus(p.status)
        setBackgroundMd(p.background_md)
      })
      .finally(() => setLoading(false))
  }, [id])

  const canSave = !!name.trim() && !!backgroundMd.trim() && name.length <= MAX_NAME

  const handleSave = async () => {
    if (!canSave) return
    setSaving(true)
    try {
      if (isEdit) {
        await projectsApi.update(id!, { name, status, background_md: backgroundMd })
        toast.success('已保存')
      } else {
        await projectsApi.create({ name, status, background_md: backgroundMd })
        toast.success('已创建')
      }
      navigate('/teacher/projects')
    } finally {
      setSaving(false)
    }
  }

  if (loading) return <div className="p-8 text-muted-foreground">加载中...</div>

  return (
    <div className="flex h-full flex-col">
      <div className="flex items-center justify-between border-b p-3">
        <Button variant="ghost" size="sm" onClick={() => navigate('/teacher/projects')}>
          ← 返回列表
        </Button>
        <div className="flex gap-2">
          <Button variant="outline" onClick={() => navigate('/teacher/projects')}>取消</Button>
          <Button onClick={handleSave} disabled={!canSave || saving}>
            {saving ? '保存中...' : '保存'}
          </Button>
        </div>
      </div>
      <div className="flex-1 overflow-auto p-4">
        <div className="mx-auto max-w-5xl space-y-4">
          <div className="grid grid-cols-2 gap-4">
            <div className="space-y-2">
              <Label htmlFor="name">名称</Label>
              <Input
                id="name"
                value={name}
                onChange={e => setName(e.target.value.slice(0, MAX_NAME))}
                placeholder="项目名称"
              />
              <div className="text-right text-xs text-muted-foreground">{name.length}/{MAX_NAME}</div>
            </div>
            <div className="space-y-2">
              <Label>状态</Label>
              <Select value={status} onValueChange={v => setStatus(v as 'draft' | 'published')}>
                <SelectTrigger>
                  <SelectValue />
                </SelectTrigger>
                <SelectContent>
                  <SelectItem value="draft">草稿</SelectItem>
                  <SelectItem value="published">已发布</SelectItem>
                </SelectContent>
              </Select>
            </div>
          </div>
          <div className="grid grid-cols-2 gap-4">
            <div className="space-y-2">
              <Label>项目背景 (Markdown)</Label>
              <textarea
                value={backgroundMd}
                onChange={e => setBackgroundMd(e.target.value)}
                placeholder={'## 实验目标\n理解...\n\n## 实验步骤\n1. ...'}
                className="min-h-96 w-full resize-y rounded-md border bg-background p-3 font-mono text-sm focus:outline-none focus:ring-1 focus:ring-ring"
              />
            </div>
            <div className="space-y-2">
              <Label>实时预览</Label>
              <div className="min-h-96 overflow-auto rounded-md border bg-background p-3">
                {backgroundMd ? (
                  <MarkdownRenderer content={backgroundMd} />
                ) : (
                  <span className="text-sm text-muted-foreground">在左侧输入 Markdown 即可看到预览</span>
                )}
              </div>
            </div>
          </div>
        </div>
      </div>
    </div>
  )
}
```
